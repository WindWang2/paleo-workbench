cmake_minimum_required(VERSION 3.24)
project(PaleoRibbonPrototype VERSION 0.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Widgets Test)
qt_add_executable(paleo-ribbon-prototype main.cpp)
target_link_libraries(paleo-ribbon-prototype PRIVATE Qt6::Widgets Qt6::Test)
if(MSVC)
  target_compile_options(paleo-ribbon-prototype PRIVATE /utf-8 /W4)
endif()
qt_add_resources(paleo-ribbon-prototype demo PREFIX "/" FILES assets/facies-demo.png)
enable_testing()
add_test(NAME ribbon.interaction COMMAND paleo-ribbon-prototype --self-test --output ${CMAKE_BINARY_DIR}/evidence)
set_tests_properties(ribbon.interaction PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
